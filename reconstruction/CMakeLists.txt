# Oracle C natif facultatif : jamais une dependance du core ou des autres
# tests. Fournir explicitement headers + bibliotheque Unicorn de meme ABI.
set(TERA_UNICORN_ROOT "" CACHE PATH "Installation native Unicorn optionnelle")
if(TERA_UNICORN_ROOT)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Le banc Unicorn local requiert une configuration native 64 bits separee")
    endif()
    file(SHA256 "${TERA_TEST_RESOURCE_DIR}/TERA.COM" tera_com_hash)
    if(NOT tera_com_hash STREQUAL "22fb836aa93f46dc2437e8b9586415d13552bfac35e8ac6647ba6157fd9b1a37")
        message(FATAL_ERROR "Empreinte TERA.COM differente de la reference auditee")
    endif()
    file(SHA256 "${TERA_TEST_RESOURCE_DIR}/TERA.000" tera_overlay_hash)
    if(NOT tera_overlay_hash STREQUAL "539d95781f9296d40ecd05698eb98255835a45cb03e60b49cf581e3ada7038f4")
        message(FATAL_ERROR "Empreinte TERA.000 differente de la reference auditee")
    endif()
    add_executable(test_binary_input_c tests/test_binary_input_c.c)
    target_include_directories(test_binary_input_c SYSTEM PRIVATE "${TERA_UNICORN_ROOT}/include")
    target_link_libraries(test_binary_input_c PRIVATE tera_engine "${TERA_UNICORN_ROOT}/lib/unicorn.dll")
    if(NOT MSVC)
        # L'API Unicorn exige la conversion callback vers void* ; pas de
        # -pedantic ici. Les avertissements stricts du core restent actifs.
        target_compile_options(test_binary_input_c PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME binary_input_boundaries COMMAND "${CMAKE_COMMAND}" -E env
        "PATH=${TERA_UNICORN_ROOT}/lib;$ENV{PATH}"
        "$<TARGET_FILE:test_binary_input_c>" "${TERA_TEST_RESOURCE_DIR}/TERA.COM" "${TERA_TEST_RESOURCE_DIR}/TERA.000")
    set_tests_properties(binary_input_boundaries PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
    add_executable(test_teramain_binary_c tests/test_campaign_c.c tests/teramain_oracle.c)
    target_compile_definitions(test_teramain_binary_c PRIVATE TERA_BINARY_CAMPAIGN=1)
    target_include_directories(test_teramain_binary_c SYSTEM PRIVATE "${TERA_UNICORN_ROOT}/include")
    target_link_libraries(test_teramain_binary_c PRIVATE tera_engine "${TERA_UNICORN_ROOT}/lib/unicorn.dll")
    if(NOT MSVC)
        target_compile_options(test_teramain_binary_c PRIVATE -Wall -Wextra -Werror)
    endif()
    foreach(seed 1986 12345)
        add_test(NAME teramain_binary_${seed} COMMAND "${CMAKE_COMMAND}" -E env
